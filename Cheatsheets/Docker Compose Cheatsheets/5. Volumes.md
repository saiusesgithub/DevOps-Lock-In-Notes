Named Volume

```yml
services:
  db:
    image: postgres
    volumes:
      - pgdata:/var/lib/postgresql/data

volumes:
  pgdata:
```

Bind Mount
```yml
volumes:
  - ./app:/app
```

Read-only
```yml
volumes:
  - ./app:/app:ro
```

Multiple Volumes
```yml
volumes:
  - pgdata:/var/lib/postgresql/data
  - ./backup:/backup
```
